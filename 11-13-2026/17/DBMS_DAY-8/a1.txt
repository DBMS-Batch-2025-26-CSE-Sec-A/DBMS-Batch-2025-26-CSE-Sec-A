CREATE TABLE Cultural_Stud (
    ROLL NUMBER(5) CONSTRAINT pk_cul_roll PRIMARY KEY,
    NAME VARCHAR2(20) CONSTRAINT uq_cul_name UNIQUE,
    DEPT VARCHAR2(10) CONSTRAINT chk_cul_dept CHECK (DEPT IN ('IT', 'CS')),
    CUL_FEES NUMBER(10, 2) CONSTRAINT chk_cul_fees CHECK (CUL_FEES >= 600)
);

CREATE TABLE Sports_Stud (
    ROLL NUMBER(5) CONSTRAINT pk_sp_roll PRIMARY KEY,
    NAME VARCHAR2(20) CONSTRAINT uq_sp_name UNIQUE,
    DEPT VARCHAR2(10) CONSTRAINT chk_sp_dept CHECK (DEPT IN ('IT', 'CS')),
    SPORTS_FEE NUMBER(10, 2) CONSTRAINT chk_sp_fee CHECK (SPORTS_FEE >= 1000)
);
