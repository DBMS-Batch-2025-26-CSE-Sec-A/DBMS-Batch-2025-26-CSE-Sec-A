CREATE TABLE Customer (
    Cust_id VARCHAR2(5) CONSTRAINT pk_customer PRIMARY KEY CONSTRAINT chk_cust_id CHECK (Cust_id LIKE 'A%'),
    Fname VARCHAR2(15) NOT NULL,
    Lname VARCHAR2(15),
    Area VARCHAR2(5) NOT NULL,
    Phone NUMBER(10)
);

CREATE TABLE Movie (
    Mv_no NUMBER(5) CONSTRAINT pk_movie PRIMARY KEY,
    Title VARCHAR2(25) NOT NULL,
    Type VARCHAR2(15) NOT NULL,
    Star VARCHAR2(10) NOT NULL,
    Price NUMBER(8, 2) NOT NULL
);

CREATE TABLE Invoice (
    Inv_no VARCHAR2(5) CONSTRAINT pk_invoice PRIMARY KEY CONSTRAINT chk_inv_no CHECK (Inv_no LIKE 'I%'),
    Mv_no NUMBER(5) CONSTRAINT fk_inv_mv REFERENCES Movie(Mv_no),
    Cust_id VARCHAR2(5) CONSTRAINT fk_inv_cust REFERENCES Customer(Cust_id),
    Issue_date DATE NOT NULL,
    Return_date DATE NOT NULL
);
