CONSTRAINTS:-
A)	Roll of both the tables is the primary key of corresponding tables.
B)	Dept should be either IT or CS.
C)	Sports_ fee should be greater or equal to 1000.
D)	Cultural_Stud should be greater than equal to 600.
E)	Name should be the unique key of the two relations.

Query:
CREATE TABLE SPORTS_STUD
(
    ROLL NUMBER(5) PRIMARY KEY,
    NAME VARCHAR2(20) UNIQUE,
    DEPT VARCHAR2(10) CHECK (DEPT IN ('IT','CS')),
    SPORTS_FEE NUMBER(6) CHECK(SPORTS_FEE>=1000)
);

CREATE TABLE CULTURAL_STUD
(
    ROLL NUMBER(5) PRIMARY KEY,
    NAME VARCHAR2(20) UNIQUE,
    DEPT VARCHAR2(10) CHECK (DEPT IN('IT','CS')),
    CUL_FEES NUMBER(6) CHECK (CUL_FEES>=600)
);

2. Display all the distinct students studying in the organization including both the departments.

Query:
SELECT NAME FROM SPORTS_STUD UNION

3. Display all the students studying in the organization including both the departments.

Query:
SELECT NAME FROM SPORTS_STUD 
UNION ALL
SELECT NAME FROM CULTURAL_STUD;


4. Display those students working in both cultural and sports group.

Query:
SELECT NAME FROM SPORTS_STUD INTERSECT
SELECT NAME FROM CULTURAL_STUD;

5. Display those students working in only sports group.

Query:
SELECT NAME FROM SPORTS_STUD MINUS
SELECT NAME FROM CULTURAL_STUD;

6. Display those students working in only cultural group.

Query:
SELECT NAME FROM CULTURAL_STUD MINUS 
SELECT NAME FROM SPORTS_STUD;

7. Display the roll and both the sports fee and cultural fee of the students working in both cultural and sports group.

Query:
SELECT S.ROLL,S.SPORTS_FEE,C.CUL_FEES FROM SPORTS_STUD S JOIN CULTURAL_STUD C ON S.ROLL=C.ROLL;

9. Display the student name, age and month of birth of all the distinct students of the organization.

Query:
SELECT NAME,TRUNC(MONTHS_BETWEEN(SYSDATE,BIRTHDAY)/12)AS AGE, TO_CHAR(BIRTHDAY, 'MONTH')AS MONTH_OF_BIRTH FROM SPORTS_STUD UNION 
SELECT NAME,TRUNC(MONTHS_BETWEEN(SYSDATE,BIRTHDAY)/12), TO_CHAR(BIRTHDAY, 'MONTH') FROM CULTURAL_STUD;

10. Display the last day of the month in which all the students were born. (Use set operators to display the result for all students).

Query:
SELECT LAST_DAY(BIRTHDAY)AS LAST_DAY FROM SPORTS_STUD UNION
SELECT LAST_DAY(BIRTHDAY) FROM CULTURAL_STUD;

11. Display the details of students from sports_stud table in the following format:
<Name> who studies in <Dept> dept pays Rs<sports_fee> for sports.

12. Display the details of all students from IT department. Display the total money earned by the organization from the sports department as well as from the cultural        department.

Query:
SELECT ROLL,NAME,DEPT,SPORTS_FEE AS FEE FROM SPORTS_STUD WHERE DEPT='IT' UNION ALL 
SELECT ROLL,NAME,DEPT,CUL_FEES FROM CULTURAL_STUD WHERE DEPT='IT';

SELECT SUM(SPORTS_FEE)AS TOTAL_SPORTS_FEE FROM SPORTS_STUD;

SELECT SUM(CUL_FEES)AS TOTAL_CULTURAL_FEE FROM CULTURAL_STUD;

13. Display the name of all the students in descending order.

Query:
SELECT NAME 
FROM
(
    SELECT NAME FROM SPORTS_STUD 
    UNION
    SELECT NAME FROM CULTURAL_STUD
)
ORDER BY NAME DESC;

15. Display the maximum, minimum and average sports_fee taken by the organization.

Query:

SELECT MAX(SPORTS_FEE) AS MAXIMUM,
       MIN(SPORTS_FEE) AS MINIMUM,
       AVG(SPORTS_FEE) AS AVERAGE
FROM SPORTS_STUD;
