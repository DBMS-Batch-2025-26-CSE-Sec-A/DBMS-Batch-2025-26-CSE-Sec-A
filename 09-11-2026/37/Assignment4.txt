    Phone NUMBER(7)

CREATE TABLE Movie
(
    Mv_no NUMBER(2) PRIMARY KEY,
    Cust_id VARCHAR2(3) NOT NULL,
    Title VARCHAR2(20) NOT NULL,
    Star VARCHAR2(2) NOT NULL,
    Price NUMBER(3) CHECK (Price Between 100 AND 250),
    FOREIGN KEY (Cust_id) REFERENCES Customer(Cust_id)
);
